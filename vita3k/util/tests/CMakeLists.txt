cmake_minimum_required(VERSION 3.20)
project(cooperative_pause_tests LANGUAGES CXX)
find_package(Threads REQUIRED)
enable_testing()
add_executable(cooperative_pause_tests cooperative_pause_tests.cpp)
target_compile_features(cooperative_pause_tests PRIVATE cxx_std_20)
target_include_directories(cooperative_pause_tests PRIVATE ../include)
target_link_libraries(cooperative_pause_tests PRIVATE Threads::Threads)
add_test(NAME cooperative_pause COMMAND cooperative_pause_tests)
set_tests_properties(cooperative_pause PROPERTIES TIMEOUT 60)
